        import json
        from typing import TypedDict
        import copy

        """
         @dataclass -> 初始化方法 配置类和实体类
         BaseModel -> 初始化方法 严格模式 json处理 -> fastapi
         TypedDict -> langgraph  -> return {}
        """

        class ImportGraphState(TypedDict):
            task_id:str                 # 标记任务的唯一标识

            local_file_path:str         # 标记输入的原文件地址(不确定类型)
            md_path:str                 # 明确的md地址 / 后续处理完图片地址
            pdf_path:str                # 明确的pdf的地址
            file_title:str              # 文件名 xx.md -> xx
            local_dir:str               # 输出文件的文件夹地址

            md_content:str              # md的内容

            is_md_read_enabled:bool     # 是md文件
            id_pdf_read_enabled:bool    # 是pdf文件

            chunk:list[dict]            # 切块的内容(没有向量)

            item_name:str               # 每个文档提取的唯一标识! 多个文档可能相同(烫金机维修/销售/售后手册)

            embeddings_content:list[dict] # 切块的内容(有向量)

        # 默认对象 模板
        graph_default_state: ImportGraphState = {
            "task_id": "",
            "is_pdf_read_enabled": False,
            "is_md_read_enabled": False,
            "local_dir": "",
            "local_file_path": "",
            "pdf_path": "",
            "md_path": "",
            "file_title": "",
            "md_content": "",
            "chunks": [],
            "item_name": "",
            "embeddings_content": [],
        }


        # 创建指定参数的对象
        def create_default_state(**kwargs) -> ImportGraphState:
            """
              创建对象可以修改参数
            :param **kwargs 传入:key -> 修改属性名 =value 修改的属性值 , key=value   接收: 字典 {key:value,key:value}
            :return:
            """

            # dict -> update -> {key:value} -> 修改原有值对应的key
            """
              深拷贝: 创建对象,会复制对象有嵌套的属性  适合有集合或者元组这类的嵌套属性  全新的对象 = copy.deepcopy(对象)
              浅拷贝: 创建对象,只会复制对象没有嵌套的属性  适合没有集合或者元组这类的嵌套属性  对象.copy() / copy.copy(对象)
            """
            state = copy.deepcopy(graph_default_state)
            state.update(kwargs)
            return state

        # 创建一个默认对象
        def get_default_state() -> ImportGraphState:
            state = copy.deepcopy(graph_default_state)
            return state

3. http超文本传输协议和requests使用用说明
   http协议
      概念: http超文本传输协议  客户端 -- 服务端
                 ----> 请求报文 -->
           客户端                  服务端
                 <--- 响应报文 ---
      数据包:
         请求报文: 4部分
              请求行:请求方式[空格]请求地址[控制]协议版本\n\r   -> 我们请求的服务器资源地址和请求方式
              请求头:key=value
                    ....                                  -> minerU 申请文件上传地址: 1. 文件名 -> 请求体 2. token -> 请求头
                    ....                                  -> 请求数据包中除了正文的数据的一些额外描述!  登录功能: 账号和密码 -> 请求体  user-agent: 什么浏览器
              请求空行:\n\r\n\r
              请求体: 请求的数据等等                         -> 请求的正文数据,本质是流, 字符 + 字节
         响应报文: 4部分
              状态行: 协议版本[空格]状态码\n\r                -> 浏览器看的,是否响应成功!!! 状态码 -> 本次请求的网络状态 200
              响应头: key=value
                      ....                                -> 服务端返回数据的额外信息 返回数据的长度 数据类型等等
                                                          -> fastapi下载一张图片:  图片 -> 响应体 || 响应头 content_disposition attachment
              响应空行:
              响应体: 响应的数据                             -> 返回的数据
      请求方式:
          请求行:GET[空格]请求地址[控制]协议版本\n\r
          请求方式有哪些? GET POST PUT DELETE
          请求方式有什么影响? 请求方式影响请求报文的数据结构! 不会影响响应报文的数据结构(固定4部分)
             GET / DELETE 轻量级:  请求报文只有两部分  请求行 请求方式[空格]请求地址[控制]协议版本\n\r 请求头  key=value
             POST / PUT   重量级:  请求报文有四部分 请求行 请求头 请求空行 请求体
          怎么选择请求方式呢?
             1. 你是否需要请求体
             2. 请求的动作 (restful)
          需求1: 上传头像  头像->字节->请求体 + 更新 -> put
          需求2: 登录功能(账号和密码)  账号和密码 -> 字符串 -> 暴漏不好 别再路径 -> 请求体 -> post
          需求3: 查询所有学生信息(没有参数)  没有参数 -> 获取 -> get
          需求4: 查询指定id的学生信息(id=x)  id -> 隐私不算 -> 字符 -> url/1 -> get

          POST/PUT传递参数: 请求体传递参数 请求体流 字符串 字节 json
          GET/DELETE传递参数: 路径传参 / ? 后传递参数  请求方式[空格]url  /user/1 ?key=value[控制]协议版本
          传递参数:
             1. 路径传递参数  /user/1                   get / delete    put/post
             2. 查询传递参数  url?key=value&key=value   get / delete    put/post
             3. 请求体字符参数  json字符串               put/post
             4. 请求体字节参数  文件...                  put/post
          路径和查询参数有啥区别?
             资源的标识 -> 路径
             资源的参数 -> ? 后
             1. 查询指定id = x 学生的信息   path  -> 对应的资源唯一标识
             2. 查询年龄大于18的学生信息     param/query
             3. 查询年龄等于18的学生信息     param/query
      响应状态码:
         状态行: 协议版本[空格]状态码\n\r
         1xx  101 102                   中继状态   不用管
         2xx  200 202                   成功状态   不用管
         3xx  304 302                   间接成功   不用管
         4xx  404 400 403 405 402 412   客户端问题        html -->   ---> fastapi  掌柜智库 -->  500  --> minerU
         5xx  500 503 501               服务端问题

      总结:
        1. 选择合适的请求方式很重要 get / delete  / put / post
        2. 选择合适的传参方式很重要  path /  param  / 请求体
        3. 理解响应状态码含义很重要  1 2 3 4 5
   requests的使用
       import requests 客户端的角色发起请求
       请求:
       response = requests.get/post/put/delete(传递参数 url="地址", params=dict ? ? , data=请求体的字节数据 , json=请求体的字符串数据 , timeout=超时时间)
            get 路径参数 requests.get(url="/user/1")
            delete param id=1参数  requests.delete(url="/user?id=1" || url="/user" ,params={"id":1})
            想minerU发起请求传递header和json  requests.post(url,headers=header,json=json)
            想指定的地址上传要解析的文件 requests.put(url,data=文件的字节数据)
       响应:
       响应状态码:  response.status_code
       响应体数据:  @property response.content() -> 响应体的字节数据 / @property text() -> 响应体的文本数据"json" / json(x,x,x) -> 响应体的json字符串 -> 转dict
        @property 1. 函数没有参数 2. 只读模型
